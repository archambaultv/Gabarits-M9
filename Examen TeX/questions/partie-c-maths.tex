\exampart{Programmation scientifique}

\begin{instructions}
  Les questions de cette partie portent sur la traduction de formules
  mathématiques en Python. Le module \codeinline{math} est importé.
\end{instructions}

% --- QCM avec des maths dans les choix -------------------------------
\begin{mcq}
  Quelle formule la fonction \codeinline{f(x, n)} calcule-t-elle ?
\begin{code}{python}
def f(x, n):
    total = 0
    for k in range(n + 1):
        total += x**k / math.factorial(k)
    return total
\end{code}
  \begin{choices}[columns=2]
    \correct $\displaystyle \sum_{k=0}^{n} \frac{x^k}{k!}$
    \wrong   $\displaystyle \sum_{k=1}^{n} \frac{x^k}{k!}$
    \wrong   $\displaystyle \sum_{k=0}^{n-1} \frac{x^k}{k!}$
    \wrong   $\displaystyle \prod_{k=0}^{n} \frac{x^k}{k!}$
  \end{choices}
  % range(n + 1) va de 0 à n inclusivement ; total += … est une somme.
\end{mcq}

% --- Question : traduire une formule ---------------------------------
\begin{question}[points=2, lines=4]
  Les racines de l'équation $ax^2 + bx + c = 0$ sont données par
  \[ x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}, \qquad a \neq 0, \quad b^2 - 4ac \geq 0. \]
  Écrivez les deux instructions Python qui affectent ces racines aux
  variables \codeinline{x1} et \codeinline{x2}.
  \begin{solution}
\begin{code}[numbers=none]{python}
x1 = (-b + math.sqrt(b**2 - 4*a*c)) / (2*a)
x2 = (-b - math.sqrt(b**2 - 4*a*c)) / (2*a)
\end{code}
    Les parenthèses autour de \codeinline{2*a} sont essentielles :
    \codeinline{y / 2*a} calcule $(y/2) \cdot a$ et non $y/(2a)$.
  \end{solution}
\end{question}

% --- Question en plusieurs parties : cadre de code, puis lignes ------
\begin{question}[points=4]
  On sait que la somme $S_n = \displaystyle\sum_{k=1}^{n} \frac{1}{k^2}$
  tend vers $\dfrac{\pi^2}{6}$ lorsque $n$ devient grand.
  \begin{subquestion}[box=4.5cm]
    Écrivez une fonction Python \codeinline{somme(n)} qui retourne $S_n$.
    \begin{solution}
\begin{code}[numbers=none]{python}
def somme(n):
    total = 0
    for k in range(1, n + 1):
        total += 1 / k**2
    return total
\end{code}
    \end{solution}
  \end{subquestion}
  \begin{subquestion}[lines=1]
    Déterminez la valeur retournée par \codeinline{somme(2)}.
    \begin{solution}
      $1 + \frac{1}{4} = 1{,}25$
    \end{solution}
  \end{subquestion}
  \begin{subquestion}[lines=2]
    Écrivez l'expression Python qui donne une approximation de $\pi$
    à l'aide de \codeinline{somme(1000)}.
    \begin{solution}
      \codeinline{math.sqrt(6 * somme(1000))}, car
      $\pi \approx \sqrt{6\,S_n}$.
    \end{solution}
  \end{subquestion}
\end{question}
